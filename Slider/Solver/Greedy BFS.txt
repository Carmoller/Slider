public class PuzzleNode
{
    public byte[,] State { get; set; } // Representing the grid
    public PuzzleNode Parent { get; set; }

    public int H { get; set; } // Heuristic value

    // High-performance priority queue comparison logic for GBFS
    // Priority 1: Lowest Heuristic (H)
    // Priority 2: Highest Depth (G) to break ties and push forward
    private int CompareNodesForGBFS(PuzzleNode x, PuzzleNode y)
    {
        int result = x.H.CompareTo(y.H);
        if (result == 0)
        {
            return y.G.CompareTo(x.G); // Inverted: higher G comes first
        }
        return result;
    }

    public PuzzleNode LocalGbfsSprint(PuzzleNode startNode, int maxNodes = 5000)
    {
        // Custom PriorityQueue (available natively in .NET 6+)
        var localOpenList = new PriorityQueue<PuzzleNode, PuzzleNode>(
            Comparer<PuzzleNode>.Create(CompareNodesForGBFS)
        );

        // Uses a hash of the board state for O(1) lookups
        var localClosedList = new HashSet<long>(); 

        // Initialize sprint
        localOpenList.Enqueue(startNode, startNode);
        localClosedList.Add(GetStateHash(startNode.State));

        int nodesExplored = 0;

        while (localOpenList.Count > 0 && nodesExplored < maxNodes)
        {
            PuzzleNode currentNode = localOpenList.Dequeue();
            nodesExplored++;

            // Goal found!
            if (currentNode.H == 0)
            {
                return currentNode; 
            }

            // Expand directions
            foreach (var move in GetValidMoves(currentNode.State))
            {
                byte[,] childState = ApplyMove(currentNode.State, move);
                long stateHash = GetStateHash(childState);

                if (!localClosedList.Contains(stateHash))
                {
                    localClosedList.Add(stateHash);

                    int childH = CalculateCompositeHeuristic(childState);
                    
                    var childNode = new PuzzleNode
                    {
                        State = childState,
                        Parent = currentNode,
                        G = currentNode.G + 1,
                        H = childH
                    };

                    localOpenList.Enqueue(childNode, childNode);
                }
            }
        }

        // Return null if the 5000 node budget was exhausted without a solution
        return null;
    }
