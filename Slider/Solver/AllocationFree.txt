public struct Entry
{
    public long Key;
    public long Value;
    public int Next; // Pointer to the next chained entry index (-1 if end of chain)
}

public class AllocationFreeMultiDictionary
{
    private readonly int[] _buckets;
    private readonly Entry[] _entries;
    private int _count;

    public AllocationFreeMultiDictionary(int capacity)
    {
        _buckets = new int[capacity];
        _entries = new Entry[capacity];
        
        // Initialize buckets with -1 to indicate empty chains
        Array.Fill(_buckets, -1);
        _count = 0;
    }

    public bool TryAdd(long key, long value)
    {
        if (_count >= _entries.Length)
        {
            return false; // Out of pre-allocated space (prevents allocation/resizing)
        }

        // Compute bucket index using an allocation-free hash
        int bucketIndex = (int)((ulong)key % (ulong)_buckets.Length);
        
        // Setup the new entry
        int newEntryIndex = _count;
        _entries[newEntryIndex].Key = key;
        _entries[newEntryIndex].Value = value;
        
        // Chain by pointing the new entry's Next to the current head of the bucket
        _entries[newEntryIndex].Next = _buckets[bucketIndex];
        
        // Make this new entry the new head of the bucket
        _buckets[bucketIndex] = newEntryIndex;
        
        _count++;
        return true;
    }

    // Allocation-free iterator using a ref struct enumerator
    public KeyChainEnumerator EnumerateValues(long key)
    {
        int bucketIndex = (int)((ulong)key % (ulong)_buckets.Length);
        return new KeyChainEnumerator(_entries, _buckets[bucketIndex], key);
    }
}

// Custom enumerator to avoid boxing allocations during foreach
public ref struct KeyChainEnumerator
{
    private readonly Entry[] _entries;
    private readonly long _targetKey;
    private int _currentEntryIndex;
    
    public long Current { get; private set; }

    public KeyChainEnumerator(Entry[] entries, int startEntryIndex, long key)
    {
        _entries = entries;
        _currentEntryIndex = startEntryIndex;
        _targetKey = key;
        Current = 0;
    }

    public bool MoveNext()
    {
        while (_currentEntryIndex != -1)
        {
            ref readonly var entry = ref _entries[_currentEntryIndex];
            _currentEntryIndex = entry.Next; // Move next pointer ahead

            if (entry.Key == _targetKey)
            {
                Current = entry.Value;
                return true;
            }
        }
        return false;
    }
}
