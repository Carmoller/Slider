private void SetupGoalBoardForScope(
    byte[,] goalBoard,
    (byte, byte)[] goalPositions,
    GoalScope scope)
{
    // Initialize all positions
    for (byte row = 0; row < _gridSize; row++)
    {
        for (byte col = 0; col < _gridSize; col++)
        {
            byte index = (byte)(row * _gridSize + col);
            if (row == _gridSize - 1 && col == _gridSize - 1)
            {
                goalPositions[0] = (row, col);
            }
            else
            {
                goalPositions[index + 1] = (row, col);
            }
        }
    }

    // Populate goal board based on scope
    switch (scope)
    {
        case GoalScope.TopRowOnly:
            SetupTopRowGoal(goalBoard, goalPositions);
            break;
        case GoalScope.LeftColumnOnly:
            SetupLeftColumnGoal(goalBoard, goalPositions);
            break;
        case GoalScope.TopRowAndLeftColumn:
            SetupTopRowAndLeftColumnGoal(goalBoard, goalPositions);
            break;
        case GoalScope.FullPuzzle:
        default:
            SetupFullPuzzleGoal(goalBoard, goalPositions);
            break;
    }
}

5. Create a scoped goal comparison method
Add this private method:
private bool BoardMeetsGoal(byte[,] currentBoard, byte[,] goalBoard, GoalScope scope)
{
    switch (scope)
    {
        case GoalScope.TopRowOnly:
            for (byte col = 0; col < _gridSize; col++)
                if (currentBoard[0, col] != goalBoard[0, col])
                    return false;
            return true;

        case GoalScope.LeftColumnOnly:
            for (byte row = 0; row < _gridSize; row++)
                if (currentBoard[row, 0] != goalBoard[row, 0])
                    return false;
            return true;

        case GoalScope.TopRowAndLeftColumn:
            for (byte col = 0; col < _gridSize; col++)
                if (currentBoard[0, col] != goalBoard[0, col])
                    return false;
            for (byte row = 1; row < _gridSize; row++)
                if (currentBoard[row, 0] != goalBoard[row, 0])
                    return false;
            return true;

        case GoalScope.FullPuzzle:
        default:
            return BoardsEqual(currentBoard, goalBoard);
    }
}
