The "10-Minute C++ Style" C# Solution
If you drop into unsafe code, you can pass around raw, unmanaged pointers inside your queues just like you would in C++. A raw pointer (byte*) is a value type. It has no object overhead, completely bypasses the GC, and can sit inside standard queues forever.csharpusing System.Runtime.InteropServices;

public unsafe class NativeQueueAllocator
{
    private byte* _rootPointer;
    private int _position = 0;
    private readonly int _totalSize;

    public NativeQueueAllocator(int sizeInBytes)
    {
        _totalSize = sizeInBytes;
        // Allocate native heap memory, exactly like C++ malloc or new byte[]
        _rootPointer = (byte*)NativeMemory.Alloc((nuint)sizeInBytes);
    }

    // Returns a raw pointer to the sub-slice (Zero GC allocation)
    public byte* Allocate(int size, out int length)
    {
        if (_position + size > _totalSize) 
            throw new OutOfMemoryException();

        byte* slice = _rootPointer + _position;
        _position += size;
        length = size;
        return slice;
    }

    public void Reset() => _position = 0;

    public void FreeAll() => NativeMemory.Free(_rootPointer);
}

public unsafe readonly struct PointerToken

    public int Length { get; }

        Address = address;
        Length = length;
    }

Use code with caution.The "10-Minute C++ Style" Struct AllocatorYou can use the exact same pointer-arithmetic strategy to allocate arrays or single blocks of structs out of a massive unmanaged buffer.csharpusing System.Runtime.InteropServices;

public unsafe class NativeStructAllocator<T> where T : unmanaged
{
    private byte* _rootPointer;
    private int _position = 0;
    private readonly int _totalSizeInBytes;

    public NativeStructAllocator(int maxStructCount)
    {
        // Calculate total size: count * size of one struct (e.g., sizeof(NetworkPacket))
        _totalSizeInBytes = maxStructCount * sizeof(T);
        _rootPointer = (byte*)NativeMemory.Alloc((nuint)_totalSizeInBytes);
    }

    // Allocate a block of 'count' structs and return a typed pointer
    public T* Allocate(int count)
    {
        int bytesRequired = count * sizeof(T);
        
        if (_position + bytesRequired > _totalSizeInBytes) 
            throw new OutOfMemoryException("Native struct buffer exhausted!");

        // Get the address, then cast it to the struct pointer type
        T* slice = (T*)(_rootPointer + _position);
        _position += bytesRequired;
        
        return slice;
    }

    public void Reset() => _position = 0;
    public void FreeAll() => NativeMemory.Free(_rootPointer);
}
Use code with caution.Enqueuing Struct PointersBecause a raw pointer (like NetworkPacket*) is just a 64-bit integer address, you can push millions of them into standard C# queues with zero GC impact.csharp// Define a token to store in the queue
public unsafe readonly struct StructToken<T> where T : unmanaged
{
    public T* Address { get; }
    public int Count { get; }

    public StructToken(T* address, int count)
    {
        Address = address;
        Count = count;
    }

    // Expose as a safe Span for read/write processing
    public Span<T> AsSpan() => new Span<T>(Address, Count);
}

// Usage inside your application
var allocator = new NativeStructAllocator<NetworkPacket>(5_000_000);
Queue<StructToken<NetworkPacket>> queue = new Queue<StructToken<NetworkPacket>>();

// 1. Allocate natively
NetworkPacket* packetPtr = allocator.Allocate(1);

// 2. Modify properties via pointer tracking (C++ style)
packetPtr->Id = 42;
packetPtr->Timestamp = DateTime.UtcNow.Ticks;

// 3. Enqueue the token (Zero GC overhead)
queue.Enqueue(new StructToken<NetworkPacket>(packetPtr, 1));
